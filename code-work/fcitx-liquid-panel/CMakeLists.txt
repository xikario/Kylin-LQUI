cmake_minimum_required(VERSION 3.16)
project(FcitxLiquidPanel LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 REQUIRED COMPONENTS Core Gui Widgets DBus Test)
add_subdirectory(../shared/liquid-glass liquid-glass)
add_subdirectory(../shared/liquid-popup liquid-popup)
find_path(FCITX_LIQUID_X11_INCLUDE_DIR X11/XKBlib.h HINTS ${UKUI_LIQUID_GL_INCLUDE_DIR})
if(NOT FCITX_LIQUID_X11_INCLUDE_DIR)
  message(FATAL_ERROR "X11 headers are required for the Caps Lock indicator monitor")
endif()
add_library(liquid-candidate STATIC src/CandidateWindow.cpp src/PinyinTones.cpp src/PanelAgent.cpp src/CandidateConfig.cpp src/SettingsDialog.cpp src/TrayMenu.cpp src/CapsLockMonitor.cpp assets/tray_icons.qrc assets/pinyin.qrc)
target_include_directories(liquid-candidate PUBLIC src)
target_include_directories(liquid-candidate SYSTEM PRIVATE ${FCITX_LIQUID_X11_INCLUDE_DIR})
target_link_libraries(liquid-candidate PUBLIC ukui-liquid-optics ukui-liquid-popup Qt5::DBus)
add_executable(fcitx-liquid-panel src/main.cpp)
set_target_properties(fcitx-liquid-panel PROPERTIES SKIP_BUILD_RPATH TRUE INSTALL_RPATH "")
target_link_libraries(fcitx-liquid-panel PRIVATE liquid-candidate)
add_executable(candidate-test tests/candidate_test.cpp)
target_link_libraries(candidate-test PRIVATE liquid-candidate Qt5::Test)
add_executable(diffusion-test ../shared/liquid-glass/tests/diffusion_test.cpp)
target_link_libraries(diffusion-test PRIVATE ukui-liquid-optics)
enable_testing()
add_test(NAME candidate COMMAND candidate-test)
add_test(NAME diffusion COMMAND diffusion-test)
set_tests_properties(candidate PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;UKUI_LIQUID_GLASS_NO_GL=1")

add_subdirectory(../ukui-liquid-desktop/shared/settings lqui-settings)
target_link_libraries(liquid-candidate PUBLIC lqui-settings-bridge)

# Legacy UI fixtures must never launch the installed central application.
get_property(LQUI_EXISTING_TESTS DIRECTORY PROPERTY TESTS)
foreach(LQUI_TEST IN LISTS LQUI_EXISTING_TESTS)
  set_property(TEST ${LQUI_TEST} APPEND PROPERTY ENVIRONMENT "LQUI_SETTINGS_LEGACY=1")
endforeach()
