cmake_minimum_required(VERSION 3.16)
project(LiquidCli VERSION 0.2.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 REQUIRED COMPONENTS Core Gui Widgets Network DBus Test)
set(LIQUID_DESKTOP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../ukui-liquid-desktop" CACHE PATH "Liquid material source")
# Reuse the existing Snell optical renderer, without building desktop apps.
set(BUILD_TESTING OFF CACHE BOOL "Shared library tests" FORCE)
add_subdirectory("${LIQUID_DESKTOP_ROOT}/shared/liquid-glass" liquid-glass)
add_subdirectory("${LIQUID_DESKTOP_ROOT}/shared/liquid-popup" liquid-popup)
add_library(cli-ui STATIC src/Settings.cpp src/Markdown.cpp src/Conversation.cpp src/AgentClient.cpp src/AgentClient.h src/CodexClient.cpp src/AcpClient.cpp src/ChatView.cpp src/Composer.cpp
  src/RegionCapture.cpp src/CaptureEditor.cpp src/Drawer.cpp src/Preferences.cpp src/SettingsHelp.cpp src/WindowIdentity.cpp assets/resources.qrc)
target_include_directories(cli-ui PUBLIC src "${LIQUID_DESKTOP_ROOT}/apps/ukui-kaishicaidan-v2/src" "${LIQUID_DESKTOP_ROOT}/shared/desktop-launch")
target_link_libraries(cli-ui PUBLIC Qt5::Widgets Qt5::Network Qt5::DBus ukui-liquid-optics ukui-liquid-popup Threads::Threads)
find_path(CLI_X11_INCLUDE_DIR X11/Xlib.h HINTS "${Qt5Core_DIR}/../../../../include" REQUIRED)
find_library(CLI_X11_LIBRARY NAMES X11 libX11.so.6 REQUIRED)
set_source_files_properties(src/WindowIdentity.cpp PROPERTIES INCLUDE_DIRECTORIES "${CLI_X11_INCLUDE_DIR}")
target_link_libraries(cli-ui PUBLIC ${CLI_X11_LIBRARY})
add_executable(liquid-cli src/main.cpp)
target_link_libraries(liquid-cli PRIVATE cli-ui)
add_executable(liquid-cli-session src/session-guard.cpp)
enable_testing()
add_executable(cli-test tests/test.cpp)
target_link_libraries(cli-test PRIVATE cli-ui Qt5::Test)
add_executable(cli-fixture tests/fixture.cpp)
target_link_libraries(cli-fixture PRIVATE Qt5::Core)
add_executable(acp-fixture tests/acp-fixture.cpp)
target_link_libraries(acp-fixture PRIVATE Qt5::Core)
find_program(DBUS_RUN_SESSION_EXECUTABLE dbus-run-session)
if(NOT DBUS_RUN_SESSION_EXECUTABLE)
  message(FATAL_ERROR "UI tests require dbus-run-session for an isolated session bus")
endif()
foreach(SCALE 1 1.5 2)
  add_test(NAME cli-ui-${SCALE} COMMAND xvfb-run -a -s "-screen 0 1920x1080x24" ${DBUS_RUN_SESSION_EXECUTABLE} -- $<TARGET_FILE:cli-test>)
  set_tests_properties(cli-ui-${SCALE} PROPERTIES TIMEOUT 120 ENVIRONMENT "QT_SCALE_FACTOR=${SCALE};QT_QPA_PLATFORMTHEME=;UKUI_LIQUID_GLASS_NO_GL=1;UKUI_FENCES_GLASS_NO_GL=1;LIQUID_CLI_FIXTURE=$<TARGET_FILE:cli-fixture>;LIQUID_ACP_FIXTURE=$<TARGET_FILE:acp-fixture>")
endforeach()
install(TARGETS liquid-cli liquid-cli-session RUNTIME DESTINATION lib/liquid-cli)

add_subdirectory("${LIQUID_DESKTOP_ROOT}/shared/settings" lqui-settings)
target_link_libraries(cli-ui PUBLIC lqui-settings-bridge)

# Legacy UI fixtures must never launch the installed central application.
get_property(LQUI_EXISTING_TESTS DIRECTORY PROPERTY TESTS)
foreach(LQUI_TEST IN LISTS LQUI_EXISTING_TESTS)
  set_property(TEST ${LQUI_TEST} APPEND PROPERTY ENVIRONMENT "LQUI_SETTINGS_LEGACY=1")
endforeach()
