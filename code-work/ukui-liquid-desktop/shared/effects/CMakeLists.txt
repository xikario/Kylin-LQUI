cmake_minimum_required(VERSION 3.16)
project(LquiEffectsPolicy LANGUAGES CXX)
find_package(Qt5 REQUIRED COMPONENTS Core)
# Header-only global effects switchboard (see EffectsPolicy.h).
if(NOT TARGET lqui-effects-policy)
    add_library(lqui-effects-policy INTERFACE)
    target_include_directories(lqui-effects-policy INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(lqui-effects-policy INTERFACE Qt5::Core)
endif()
if(BUILD_TESTING AND NOT TARGET lqui-effects-policy-test)
    add_executable(lqui-effects-policy-test tests/effects_policy_test.cpp)
    target_link_libraries(lqui-effects-policy-test PRIVATE lqui-effects-policy)
    add_test(NAME lqui-effects-policy COMMAND lqui-effects-policy-test)
    set_tests_properties(lqui-effects-policy PROPERTIES TIMEOUT 15)
endif()
if(BUILD_TESTING AND NOT TARGET lqui-effects-probe)
    add_executable(lqui-effects-probe tests/effects_probe.cpp)
    target_link_libraries(lqui-effects-probe PRIVATE lqui-effects-policy)
endif()
